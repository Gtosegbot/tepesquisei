const mongoose = require('mongoose');
const bcrypt = require('bcryptjs');

const UserSchema = new mongoose.Schema({
  name: {
    type: String,
    required: true
  },
  email: {
    type: String,
    required: true,
    unique: true
  },
  password: {
    type: String,
    required: function() {
      return this.googleId ? false : true;
    }
  },
  googleId: {
    type: String
  },
  role: {
    type: String,
    enum: ['owner', 'researcher', 'admin', 'superadmin'],
    default: 'owner'
  },
  credits: {
    type: Number,
    default: 0
  },
  createdAt: {
    type: Date,
    default: Date.now
  }
});

// Hash password before saving
UserSchema.pre('save', async function(next) {
  if (!this.isModified('password') || !this.password) return next();
  
  const salt = await bcrypt.genSalt(10);
  this.password = await bcrypt.hash(this.password, salt);
  next();
});

// Method to check password validity
UserSchema.methods.matchPassword = async function(enteredPassword) {
  if (!this.password) return false;
  return await bcrypt.compare(enteredPassword, this.password);
};

module.exports = mongoose.model('User', UserSchema);

// Register User
exports.register = async (req, res) => {
  try {
    const { name, email, password } = req.body;

    // Check if user exists
    const userExists = await User.findOne({ email });
    if (userExists) {
      return res.status(400).json({ message: 'User already exists' });
    }

    // Create user
    const user = await User.create({
      name,
      email,
      password
    });

    if (user) {
      res.status(201).json({
        _id: user._id,
        name: user.name,
        email: user.email,
        role: user.role,
        credits: user.credits,
        token: generateToken(user._id)
      });
    } else {
      res.status(400).json({ message: 'Invalid user data' });
    }
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};

// Google Login
exports.googleLogin = async (req, res) => {
  try {
    const { idToken } = req.body;
    
    const ticket = await client.verifyIdToken({
      idToken,
      audience: process.env.GOOGLE_CLIENT_ID
    });
    
    const { email_verified, name, email, sub: googleId } = ticket.getPayload();
    
    if (email_verified) {
      let user = await User.findOne({ email });
      
      if (!user) {
        // Create new user
        user = await User.create({
          name,
          email,
          googleId
        });
      } else {
        // Update Google ID if user already exists but logged in with Google for the first time
        if (!user.googleId) {
          user.googleId = googleId;
          await user.save();
        }
      }
      
      res.json({
        _id: user._id,
        name: user.name,
        email: user.email,
        role: user.role,
        credits: user.credits,
        token: generateToken(user._id)
      });
    } else {
      res.status(400).json({ message: 'Google login failed. Email not verified.' });
    }
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};

// Set demographic filters
exports.setDemographicFilters = async (req, res) => {
  try {
    const { surveyId } = req.params;
    const { gender, ageRanges, locations } = req.body;

    const survey = await Survey.findOne({ 
      surveyId, 
      owner: req.user._id 
    });

    if (!survey) {
      return res.status(404).json({ message: 'Survey not found' });
    }

    // Validate demographic filters
    // Check if gender distribution totals 100%
    if (gender) {
      const totalGenderDist = Object.values(gender.distribution).reduce((sum, val) => sum + Number(val), 0);
      if (Math.round(totalGenderDist) !== 100) {
        return res.status(400).json({ message: 'Gender distribution must total 100%' });
      }
      survey.demographicFilters.gender = gender;
    }

    // Check if age range distribution totals 100%
    if (ageRanges) {
      const totalAgeDist = Object.values(ageRanges.distribution).reduce((sum, val) => sum + Number(val), 0);
      if (Math.round(totalAgeDist) !== 100) {
        return res.status(400).json({ message: 'Age range distribution must total 100%' });
      }
      survey.demographicFilters.ageRanges = ageRanges;
    }

    // Check if location distribution totals 100%
    if (locations) {
      const totalLocationDist = Object.values(locations.distribution).reduce((sum, val) => sum + Number(val), 0);
      if (Math.round(totalLocationDist) !== 100) {
        return res.status(400).json({ message: 'Location distribution must total 100%' });
      }
      survey.demographicFilters.locations = locations;
    }

    // Create demographic questions
    if (gender && gender.options && gender.options.length > 0) {
      const genderQuestion = await Question.create({
        survey: survey._id,
        text: 'Qual é o seu sexo?',
        type: 'single',
        required: true,
        options: gender.options.map(g => ({ text: g, value: g })),
        order: 1,
        isDemographic: true,
        demographicType: 'gender'
      });
      
      survey.questions.push(genderQuestion._id);
    }

    if (ageRanges && ageRanges.options && ageRanges.options.length > 0) {
      const ageQuestion = await Question.create({
        survey: survey._id,
        text: 'Qual é a sua faixa etária?',
        type: 'single',
        required: true,
        options: ageRanges.options.map(a => ({ text: `${a.min} - ${a.max} anos`, value: `${a.min}-${a.max}` })),
        order: 2,
        isDemographic: true,
        demographicType: 'age'
      });
      
      survey.questions.push(ageQuestion._id);
    }

    if (locations) {
      const locationQuestion = await Question.create({
        survey: survey._id,
        text: 'Qual é a sua localização?',
        type: 'single',
        required: true,
        options: locations.regions ? 
                 locations.regions.map(r => ({ text: r, value: r })) : 
                 locations.cities.map(c => ({ text: c, value: c })),
        order: 3,
        isDemographic: true,
        demographicType: 'location'
      });
      
      survey.questions.push(locationQuestion._id);
    }

    await survey.save();
    res.json(survey);
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};

// Activate survey
exports.activateSurvey = async (req, res) => {
  try {
    const { surveyId } = req.params;

    const survey = await Survey.findOne({ 
      surveyId, 
      owner: req.user._id 
    });

    if (!survey) {
      return res.status(404).json({ message: 'Survey not found' });
    }

    // Check if survey has demographic questions
    if (!survey.demographicFilters.gender || 
        !survey.demographicFilters.ageRanges || 
        !survey.demographicFilters.locations) {
      return res.status(400).json({ 
        message: 'Survey must have demographic filters (gender, age ranges, and locations) before activation' 
      });
    }

    // Check if survey has at least one non-demographic question
    const nonDemographicQuestions = await Question.find({
      survey: survey._id,
      isDemographic: false
    });

    if (nonDemographicQuestions.length === 0) {
      return res.status(400).json({
        message: 'Survey must have at least one non-demographic question'
      });
    }

    survey.status = 'active';
    await survey.save();

    res.json(survey);
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};

      return res.status(400).json({ message: 'Demographic information is required' });
    }

    // Check if all required questions are answered
    const requiredQuestionIds = survey.questions
      .filter(q => q.required)
      .map(q => q._id.toString());

    const answeredQuestionIds = answers.map(a => a.question.toString());
    
    const missingRequiredQuestions = requiredQuestionIds.filter(
      qId => !answeredQuestionIds.includes(qId)
    );

    if (missingRequiredQuestions.length > 0) {
      return res.status(400).json({ 
        message: 'All required questions must be answered',
        missingQuestions: missingRequiredQuestions 
      });
    }

    // Process audio recording if provided
    let audioUrl = null;
    if (audioRecording) {
      const uploadResult = await uploadToCloudinary(audioRecording, 'audio');
      audioUrl = uploadResult.url;
    }

    // Create response
    const response = new Response({
      survey: survey._id,
      answers,
      demographic,
      responseMethod: req.body.responseMethod || 'web',
      ipAddress: req.ip,
      deviceInfo: req.headers['user-agent'],
      completedAt: new Date(),
      ...(req.user && { respondent: req.user._id }),
      ...(audioUrl && { audioRecording: { url: audioUrl } }),
      ...(coordinates && { 
        demographic: {
          ...demographic,
          location: {
            ...demographic.location,
            coordinates
          }
        }
      })
    });

    await response.save();

    // Update survey response count
    survey.responsesCount += 1;
    await survey.save();

    res.status(201).json({

      responseId: response._id

  } catch (error) {

exports.getSurveyStatistics = async (req, res) => {
  try {
    const { surveyId } = req.params;
    const survey = await Survey.findOne({ surveyId });
    
    if (!survey) {
      return res.status(404).json({ message: 'Survey not found' });
    }

    // Check if user is authorized to view statistics
    if (survey.owner.toString() !== req.user._id.toString() && 
        !survey.researchers.includes(req.user._id)) {
      return res.status(403).json({ message: 'Not authorized to view these statistics' });
    }

    const responses = await Response.find({ survey: survey._id });
    
    // Calculate statistics
    const demographicStats = {
      gender: {},
      ageRanges: {},
      locations: {}
    };

    // Process gender stats
    responses.forEach(response => {
      const gender = response.demographic.gender;
      demographicStats.gender[gender] = (demographicStats.gender[gender] || 0) + 1;
      
      const ageRange = `${response.demographic.ageRange.min}-${response.demographic.ageRange.max}`;
      demographicStats.ageRanges[ageRange] = (demographicStats.ageRanges[ageRange] || 0) + 1;
      
      const location = response.demographic.location.region || 
                       response.demographic.location.city || 
                       response.demographic.location.state;
      demographicStats.locations[location] = (demographicStats.locations[location] || 0) + 1;
    });

    // Process question stats
    const questions = await Question.find({ survey: survey._id });
    const questionStats = {};
    
    for (const question of questions) {
      if (question.type === 'open') continue; // Skip open-ended questions

      questionStats[question._id] = {
        questionText: question.text,
        type: question.type,
        options: {}
      };
      
      // Initialize options
      question.options.forEach(option => {
        questionStats[question._id].options[option.value] = 0;
      });
    }

    // Count answers
    responses.forEach(response => {
      response.answers.forEach(answer => {
        if (!questionStats[answer.question]) return;
        
        if (Array.isArray(answer.selectedOptions)) {
          answer.selectedOptions.forEach(option => {
            questionStats[answer.question].options[option] = 
              (questionStats[answer.question].options[option] || 0) + 1;
          });
        } else if (answer.selectedOptions) {
          questionStats[answer.question].options[answer.selectedOptions] = 
            (questionStats[answer.question].options[answer.selectedOptions] || 0) + 1;
        }
      });
    });

    res.json({
      total: responses.length,
      goal: survey.responseGoal,
      demographicStats,
      questionStats
    });
  } catch (error) {
    res.status(500).json({ message: error.message });
  }

    const userId = req.user._id;

    // Verify payment status
    const status = await checkPixPaymentStatus(transactionId);
    
    if (status === 'paid') {
      // Find the credit record
      const creditRecord = await Credit.findOne({

        user: userId,
        transactionType: 'purchase'
      });

      if (!creditRecord) {
        return res.status(404).json({ message: 'Transaction not found' });
      }

      // Check if already processed
      if (creditRecord.processed) {
        return res.json({
          status: 'paid',
          message: 'Credits already added to your account',
          amount: creditRecord.amount
        });
      }

      // Add credits to user
      const user = await User.findById(userId);
      user.credits += creditRecord.amount;
      await user.save();

      // Mark as processed
      creditRecord.processed = true;
      await creditRecord.save();

      res.json({
        status: 'paid',
        message: 'Payment confirmed, credits added to your account',
        amount: creditRecord.amount,
        newCreditBalance: user.credits
      });
    } else {
      res.json({
        status,
        message: status === 'pending' ? 'Payment pending' : 'Payment failed'
      });

exports.useCredits = async (req, res) => {
  try {
    const { type, quantity } = req.body;
    const userId = req.user._id;
    
    // Define credit cost per type
    const costs = {
      email: 0.5,
      sms: 1.0,
      whatsapp: 3.0
    };
    
    if (!costs[type]) {
      return res.status(400).json({ message: 'Invalid message type' });
    }
    
    const totalCost = costs[type] * quantity;
    
    // Check if user has enough credits
    const user = await User.findById(userId);
    
    if (user.credits < totalCost) {
      return res.status(400).json({ 
        message: 'Insufficient credits',
        required: totalCost,
        available: user.credits
      });
    }
    
    // Deduct credits
    user.credits -= totalCost;
    await user.save();
    
    // Log transaction
    await Credit.create({
      user: userId,
      amount: -totalCost,
      transactionType: 'usage',
      usageType: type,
      description: `Used ${totalCost} credits for sending ${quantity} ${type} messages`
    });
    
    res.json({
      message: `Successfully used ${totalCost} credits for ${quantity} ${type} messages`,
      newCreditBalance: user.credits
    });
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};

      }

  }, []);

  useEffect(() => {
    if (selectedState) {

        try {
          const response = await fetch(`https://servicodados.ibge.gov.br/api/v1/localidades/estados/${selectedState}/municipios`);
          const data = await response.json();
          setCities(data.sort((a, b) => a.nome.localeCompare(b.nome)));

    setSurveyData({

      {/* Step 3: Age Distribution */}
      {currentStep === 3 && (
        <div className="bg-white rounded-lg shadow-md p-6">
          <h2 className="text-xl font-semibold mb-4">Distribuição por Faixa Etária</h2>
          <p className="mb-4 text-gray-600">Defina qual a porcentagem ideal de cada faixa etária para sua pesquisa. O total deve ser 100%.</p>
          
          <div className="mb-6">
            {surveyData.demographicFilters.ageRanges.options.map(age => (
              <div key={`${age.min}-${age.max}`} className="flex items-center mb-2">
                <label className="w-1/3 text-gray-700">{age.min}-{age.max} anos</label>
                <input 
                  type="number" 
                  className="w-20 p-2 border rounded"
                  min="0"
                  max="100"
                  value={surveyData.demographicFilters.ageRanges.distribution[`${age.min}-${age.max}`] || 0}
                  onChange={(e) => handleAgeDistributionChange(`${age.min}-${age.max}`, e.target.value)}
                />
                <span className="ml-2">%</span>
              </div>
            ))}
            
            <div className="mt-4 font-semibold">
              Total: {Object.values(surveyData.demographicFilters.ageRanges.distribution)
                .reduce((sum, val) => sum + val, 0)}%
            </div>
          </div>
          
          <div className="flex justify-between">
            <button 
              className="bg-gray-300 text-gray-700 px-4 py-2 rounded hover:bg-gray-400"
              onClick={handlePrevStep}
            >
              Voltar
            </button>
            <button 
              className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
              onClick={handleNextStep}
            >
              Próximo
            </button>
          </div>
        </div>
      )}
      
      {/* Step 4: Location Selection and Distribution */}
      {currentStep === 4 && (
        <div className="bg-white rounded-lg shadow-md p-6">
          <h2 className="text-xl font-semibold mb-4">Seleção de Localização</h2>
          
          <div className="mb-4">
            <label className="block text-gray-700 mb-2">Estado*</label>
            <select 
              className="w-full p-2 border rounded"
              value={selectedState}
              onChange={handleStateChange}
              required
            >
              <option value="">Selecione um estado</option>
              {states.map(state => (
                <option key={state.id} value={state.id}>{state.nome}</option>
              ))}
            </select>
          </div>
          
          {selectedState && (
            <div className="mb-6">
              <label className="block text-gray-700 mb-2">Cidades*</label>
              <select 
                className="w-full p-2 border rounded"
                multiple
                size={5}
                value={selectedCities}
                onChange={handleCitySelection}
                required
              >
                {cities.map(city => (
                  <option key={city.id} value={city.id}>{city.nome}</option>
                ))}
              </select>
              <p className="text-sm text-gray-500 mt-1">
                Segure Ctrl (ou Cmd no Mac) para selecionar múltiplas cidades
              </p>
            </div>
          )}
          
          {surveyData.demographicFilters.locations.cities.length > 0 && (
            <div className="mb-6">
              <h3 className="font-semibold mb-2">Distribuição por Localização</h3>
              <p className="mb-4 text-gray-600">Defina qual a porcentagem ideal de cada cidade para sua pesquisa. O total deve ser 100%.</p>
              
              {surveyData.demographicFilters.locations.cities.map(city => (
                <div key={city} className="flex items-center mb-2">
                  <label className="w-1/3 text-gray-700">{city}</label>
                  <input 
                    type="number" 
                    className="w-20 p-2 border rounded"
                    min="0"
                    max="100"
                    value={surveyData.demographicFilters.locations.distribution[city] || 0}
                    onChange={(e) => handleLocationDistributionChange(city, e.target.value)}
                  />
                  <span className="ml-2">%</span>
                </div>
              ))}
              
              <div className="mt-4 font-semibold">
                Total: {Object.values(surveyData.demographicFilters.locations.distribution)
                  .reduce((sum, val) => sum + val, 0)}%
              </div>
            </div>
          )}
          
          <div className="flex justify-between">
            <button 
              className="bg-gray-300 text-gray-700 px-4 py-2 rounded hover:bg-gray-400"
              onClick={handlePrevStep}
            >
              Voltar
            </button>
            <button 
              className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
              onClick={handleNextStep}
              disabled={!selectedState || selectedCities.length === 0}
            >
              Próximo
            </button>
          </div>
        </div>
      )}
      
      {/* Step 5: Questions */}
      {currentStep === 5 && (
        <div className="bg-white rounded-lg shadow-md p-6">
          <h2 className="text-xl font-semibold mb-4">Adicionar Perguntas</h2>
          
          <div className="mb-6">
            <h3 className="font-semibold mb-2">Perguntas Adicionadas ({surveyData.questions.length})</h3>
            {surveyData.questions.length === 0 ? (
              <p className="text-gray-500">Nenhuma pergunta adicionada ainda.</p>
            ) : (
              <ul className="border rounded divide-y">
                {surveyData.questions.map((q, idx) => (
                  <li key={idx} className="p-3">
                    <div className="font-medium">{idx + 1}. {q.text}</div>
                    <div className="text-sm text-gray-500 mt-1">
                      Tipo: {q.type === 'single' ? 'Escolha única' : 
                             q.type === 'multiple' ? 'Múltipla escolha' : 
                             q.type === 'open' ? 'Resposta aberta' : 
                             q.type === 'single_with_open' ? 'Escolha única com resposta aberta' : 
                             'Múltipla escolha com resposta aberta'}
                    </div>
                  </li>
                ))}
              </ul>
            )}
          </div>
          
          <div className="border-t pt-4 mb-6">
            <h3 className="font-semibold mb-4">Nova Pergunta</h3>
            
            <div className="mb-4">
              <label className="block text-gray-700 mb-2">Texto da Pergunta*</label>
              <input 
                type="text" 
                className="w-full p-2 border rounded"
                value={newQuestion.text}
                onChange={(e) => setNewQuestion({ ...newQuestion, text: e.target.value })}
                required
              />
            </div>
            
            <div className="mb-4">
              <label className="block text-gray-700 mb-2">Tipo de Pergunta*</label>
              <select 
                className="w-full p-2 border rounded"
                value={newQuestion.type}
                onChange={(e) => setNewQuestion({ ...newQuestion, type: e.target.value })}
                required
              >
                <option value="single">Escolha única</option>
                <option value="multiple">Múltipla escolha</option>
                <option value="open">Resposta aberta</option>
                <option value="single_with_open">Escolha única com resposta aberta</option>
                <option value="multiple_with_open">Múltipla escolha com resposta aberta</option>
              </select>
            </div>
            
            <div className="mb-4">
              <div className="flex items-center mb-2">
                <input 
                  type="checkbox" 
                  id="required"
                  className="mr-2"
                  checked={newQuestion.required}
                  onChange={(e) => setNewQuestion({ ...newQuestion, required: e.target.checked })}
                />
                <label htmlFor="required" className="text-gray-700">Resposta obrigatória</label>
              </div>
            </div>
            
            {newQuestion.type !== 'open' && (
              <div className="mb-4">
                <label className="block text-gray-700 mb-2">Opções de Resposta*</label>
                {newQuestion.options.map((option, index) => (
                  <div key={index} className="flex items-center mb-2">
                    <input 
                      type="text" 
                      className="flex-1 p-2 border rounded"
                      value={option}
                      onChange={(e) => handleOptionChange(index, e.target.value)}
                      placeholder={`Opção ${index + 1}`}
                      required
                    />
                    {index > 0 && (
                      <button 
                        type="button"
                        className="ml-2 text-red-500"
                        onClick={() => handleRemoveOption(index)}
                      >
                        <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={1.5} stroke="currentColor" className="w-5 h-5">
                          <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
                        </svg>
                      </button>
                    )}
                  </div>
                ))}
                <button 
                  type="button"
                  className="mt-2 text-blue-500 flex items-center"
                  onClick={handleAddOption}
                >
                  <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={1.5} stroke="currentColor" className="w-5 h-5 mr-1">
                    <path strokeLinecap="round" strokeLinejoin="round" d="M12 4.5v15m7.5-7.5h-15" />
                  </svg>
                  Adicionar opção
                </button>
              </div>
            )}
            
            <button 
              type="button"
              className="bg-green-500 text-white px-4 py-2 rounded hover:bg-green-600"
              onClick={handleAddQuestion}
            >
              Adicionar Pergunta
            </button>
          </div>
          
          <div className="flex justify-between">
            <button 
              className="bg-gray-300 text-gray-700 px-4 py-2 rounded hover:bg-gray-400"
              onClick={handlePrevStep}
            >
              Voltar
            </button>
            <button 
              className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
              onClick={handleNextStep}
              disabled={surveyData.questions.length === 0}
            >
              Próximo
            </button>
          </div>
        </div>
      )}
      
      {/* Step 6: Review and Create */}
      {currentStep === 6 && (
        <div className="bg-white rounded-lg shadow-md p-6">
          <h2 className="text-xl font-semibold mb-4">Revisar e Criar Pesquisa</h2>
          
          <div className="mb-6">
            <h3 className="font-semibold mb-2">Informações Básicas</h3>
            <p><strong>Título:</strong> {surveyData.title}</p>
            {surveyData.description && <p><strong>Descrição:</strong> {surveyData.description}</p>}
          </div>
          
          <div className="mb-6">
            <h3 className="font-semibold mb-2">Filtros Demográficos</h3>
            <div className="mb-2">
              <strong>Gênero:</strong>
              <ul className="list-disc ml-6">
                {Object.entries(surveyData.demographicFilters.gender.distribution).map(([gender, percentage]) => (
                  <li key={gender}>{gender}: {percentage}%</li>
                ))}
              </ul>
            </div>
            <div className="mb-2">
              <strong>Faixa Etária:</strong>
              <ul className="list-disc ml-6">
                {Object.entries(surveyData.demographicFilters.ageRanges.distribution).map(([age, percentage]) => (
                  <li key={age}>{age} anos: {percentage}%</li>
                ))}
              </ul>
            </div>
            <div>
              <strong>Localização:</strong>
              <p>Estado: {surveyData.demographicFilters.locations.state}</p>
              <ul className="list-disc ml-6">
                {Object.entries(surveyData.demographicFilters.locations.distribution).map(([location, percentage]) => (
                  <li key={location}>{location}: {percentage}%</li>
                ))}
              </ul>
            </div>
          </div>
          
          <div className="mb-6">
            <h3 className="font-semibold mb-2">Perguntas ({surveyData.questions.length})</h3>
            <ul className="list-decimal ml-6">
              {surveyData.questions.map((q, idx) => (
                <li key={idx} className="mb-2">
                  <div className="font-medium">{q.text}</div>
                  <div className="text-sm text-gray-500">
                    Tipo: {q.type === 'single' ? 'Escolha única' : 
                          q.type === 'multiple' ? 'Múltipla escolha' : 
                          q.type === 'open' ? 'Resposta aberta' : 
                          q.type === 'single_with_open' ? 'Escolha única com resposta aberta' : 
                          'Múltipla escolha com resposta aberta'}
                    {q.type !== 'open' && (
                      <div>
                        Opções: {q.options.filter(o => o.trim()).join(', ')}
                      </div>
                    )}
                  </div>
                </li>
              ))}
            </ul>
          </div>
          
          <div className="flex justify-between">
            <button 
              className="bg-gray-300 text-gray-700 px-4 py-2 rounded hover:bg-gray-400"
              onClick={handlePrevStep}
            >
              Voltar
            </button>
            <button 
              className="bg-green-500 text-white px-4 py-2 rounded hover:bg-green-600"
              onClick={handleCreateSurvey}
              disabled={loading}
            >
              {loading ? 'Criando...' : 'Criar Pesquisa'}
            </button>
          </div>
        </div>
      )}
    </div>
  );
};

export default SurveyCreate;

SurveyResponse.jsx

jsx
import React, { useState, useEffect, useRef } from 'react';
import { useParams, useNavigate } from 'react-router-dom';
import { toast } from 'react-toastify';
import api from '../../utils/api';
import Loader from '../common/Loader';

const SurveyResponse = () => {
  const { surveyId } = useParams();
  const navigate = useNavigate();
  const [survey, setSurvey] = useState(null);
  const [loading, setLoading] = useState(true);
  const [submitting, setSubmitting] = useState(false);
  const [currentQuestionIndex, setCurrentQuestionIndex] = useState(0);
  const [answers, setAnswers] = useState([]);
  const [demographic, setDemographic] = useState({
    gender: '',
    ageRange: { min: 0, max: 0 },
    location: {
      state: '',
      city: '',
      region: '',
      coordinates: { latitude: null, longitude: null }
    }
  });
  const [recording, setRecording] = useState(false);
  const [audioBlob, setAudioBlob] = useState(null);
  const mediaRecorderRef = useRef(null);
  const audioChunksRef = useRef([]);
  const [currentPosition, setCurrentPosition] = useState(null);

  // Get survey data
  useEffect(() => {
    const fetchSurvey = async () => {
      try {
        const response = await api.get(`/api/surveys/${surveyId}`);
        setSurvey(response.data);
        
        // Initialize answers array with empty values for each question
        setAnswers(response.data.questions.map(() => ({
          question: '',
          selectedOptions: [],
          openResponse: ''
        })));
        
        setLoading(false);
      } catch (error) {
        console.error('Error fetching survey:', error);
        toast.error('Erro ao carregar a pesquisa');
        setLoading(false);
      }
    };
    
    fetchSurvey();
  }, [surveyId]);

  // Get geolocation
  useEffect(() => {
    if (navigator.geolocation) {
      navigator.geolocation.getCurrentPosition(
        (position) => {
          setCurrentPosition({
            latitude: position.coords.latitude,
            longitude: position.coords.longitude
          });
          
          // Update demographic with coordinates
          setDemographic(prev => ({
            ...prev,
            location: {
              ...prev.location,
              coordinates: {
                latitude: position.coords.latitude,
                longitude: position.coords.longitude
              }
            }
          }));
        },
        (error) => {
          console.error('Error getting location:', error);
        }
      );
    }
  }, []);

  // Start recording on first render
  useEffect(() => {
    if (!loading && survey) {
      startRecording();
    }
    
    return () => {
      // Cleanup: stop recording if component unmounts
      if (mediaRecorderRef.current && recording) {
        stopRecording();
      }
    };
  }, [loading, survey]);

  const startRecording = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      mediaRecorderRef.current = new MediaRecorder(stream);
      
      mediaRecorderRef.current.ondataavailable = (event) => {
        if (event.data.size > 0) {
          audioChunksRef.current.push(event.data);
        }
      };
      
      mediaRecorderRef.current.onstop = () => {
        const audioBlob = new Blob(audioChunksRef.current, { type: 'audio/wav' });
        setAudioBlob(audioBlob);
      };
      
      audioChunksRef.current = [];
      mediaRecorderRef.current.start();
      setRecording(true);
    } catch (error) {
      console.error('Error starting recording:', error);
      toast.error('Não foi possível iniciar a gravação de áudio');
    }
  };

  const stopRecording = () => {
    if (mediaRecorderRef.current && recording) {
      mediaRecorderRef.current.stop();
      setRecording(false);
      
      // Stop all audio tracks
      mediaRecorderRef.current.stream.getTracks().forEach(track => track.stop());
    }
  };

  const handleSingleChoice = (questionIndex, optionValue) => {
    const updatedAnswers = [...answers];
    updatedAnswers[questionIndex] = {
      ...updatedAnswers[questionIndex],
      question: survey.questions[questionIndex]._id,
      selectedOptions: [optionValue]
    };
    setAnswers(updatedAnswers);
  };

  const handleMultipleChoice = (questionIndex, optionValue) => {
    const updatedAnswers = [...answers];
    const currentSelectedOptions = [...updatedAnswers[questionIndex].selectedOptions];
    
    // Toggle selection
    if (currentSelectedOptions.includes(optionValue)) {
      updatedAnswers[questionIndex] = {
        ...updatedAnswers[questionIndex],
        question: survey.questions[questionIndex]._id,
        selectedOptions: currentSelectedOptions.filter(option => option !== optionValue)
      };
    } else {
      updatedAnswers[questionIndex] = {
        ...updatedAnswers[questionIndex],
        question: survey.questions[questionIndex]._id,
        selectedOptions: [...currentSelectedOptions, optionValue]
      };
    }
    
    setAnswers(updatedAnswers);
  };

  const handleOpenResponse = (questionIndex, value) => {
    const updatedAnswers = [...answers];
    updatedAnswers[questionIndex] = {
      ...updatedAnswers[questionIndex],
      question: survey.questions[questionIndex]._id,
      openResponse: value
    };
    setAnswers(updatedAnswers);
  };

  const handleNextQuestion = () => {
    const currentQuestion = survey.questions[currentQuestionIndex];
    
    // Validate current question
    if (currentQuestion.required) {
      const currentAnswer = answers[currentQuestionIndex];
      
      if (currentQuestion.type === 'open' && !currentAnswer.openResponse) {
        toast.error('Por favor, responda a pergunta antes de continuar');
        return;
      }
      
      if (['single', 'single_with_open'].includes(currentQuestion.type) && 
          currentAnswer.selectedOptions.length === 0) {
        toast.error('Por favor, selecione uma opção antes de continuar');
        return;
      }
      
      if (['multiple', 'multiple_with_open'].includes(currentQuestion.type) && 
          currentAnswer.selectedOptions.length === 0) {
        toast.error('Por favor, selecione pelo menos uma opção antes de continuar');
        return;
      }
      
      if (['single_with_open', 'multiple_with_open'].includes(currentQuestion.type) && 
          !currentAnswer.openResponse) {
        toast.error('Por favor, preencha a resposta descritiva antes de continuar');
        return;
      }
    }
    
    // Handle demographic questions
    if (currentQuestion.isDemographic) {
      if (currentQuestion.demographicType === 'gender') {
        setDemographic({
          ...demographic,
          gender: answers[currentQuestionIndex].selectedOptions[0]
        });
      } else if (currentQuestion.demographicType === 'age') {
        const ageRange = answers[currentQuestionIndex].selectedOptions[0].split('-');
        setDemographic({
          ...demographic,
          ageRange: {
            min: parseInt(ageRange[0]),
            max: parseInt(ageRange[1])
          }
        });
      } else if (currentQuestion.demographicType === 'location') {
        setDemographic({
          ...demographic,
          location: {
            ...demographic.location,
            region: answers[currentQuestionIndex].selectedOptions[0],
            state: survey.demographicFilters.locations.state,
            city: survey.demographicFilters.locations.cities.length === 1 
                ? survey.demographicFilters.locations.cities[0]
                : answers[currentQuestionIndex].selectedOptions[0]
          }
        });
      }
    }
    
    // Go to next question
    if (currentQuestionIndex < survey.questions.length - 1) {
      setCurrentQuestionIndex(currentQuestionIndex + 1);
    } else {
      // Final validation before submission
      const requiredQuestions = survey.questions.filter(q => q.required);
      const unansweredQuestions = requiredQuestions.filter((q, index) => {
        const answer = answers[index];
        
        if (q.type === 'open') {
          return !answer.openResponse;
        } else if (['single', 'single_with_open'].includes(q.type)) {
          return answer.selectedOptions.length === 0;
        } else if (['multiple', 'multiple_with_open'].includes(q.type)) {
          return answer.selectedOptions.length === 0;
        }
        
        return false;
      });
      
      if (unansweredQuestions.length > 0) {
        toast.error('Por favor, responda todas as perguntas obrigatórias antes de finalizar');
        return;
      }
      
      handleSubmit();
    }
  };

  const handlePrevQuestion = () => {
    if (currentQuestionIndex > 0) {
      setCurrentQuestionIndex(currentQuestionIndex - 1);
    }
  };

  const handleSubmit = async () => {
    // Stop recording
    stopRecording();
    
    setSubmitting(true);
    
    try {
      // Prepare audio data if available
      let audioData = null;
      if (audioBlob) {
        const reader = new FileReader();
        audioData = await new Promise((resolve) => {
          reader.onloadend = () => resolve(reader.result);
          reader.readAsDataURL(audioBlob);
        });
      }
      
      // Submit response
      await api.post(`/api/surveys/${surveyId}/responses`, {
        answers,
        demographic,
        audioRecording: audioData,
        coordinates: currentPosition,
        responseMethod: 'web'
      });
      
      toast.success('Resposta enviada com sucesso!');
      navigate('/thank-you');
    } catch (error) {
      console.error('Error submitting response:', error);
      toast.error('Erro ao enviar resposta. Por favor, tente novamente.');
    } finally {
      setSubmitting(false);
    }
  };

  if (loading) {
    return <Loader />;
  }

  if (!survey) {
    return (
      <div className="max-w-4xl mx-auto p-6 text-center">
        <h1 className="text-2xl font-bold text-red-500">Pesquisa não encontrada</h1>
        <p className="mt-4">A pesquisa que você está procurando não existe ou foi removida.</p>
      </div>
    );
  }

  const currentQuestion = survey.questions[currentQuestionIndex];

  return (
    <div className="max-w-4xl mx-auto p-6">
      <div className="bg-white rounded-lg shadow-md p-6">
        <div className="mb-4">
          <h1 className="text-2xl font-bold mb-2">{survey.title}</h1>
          {survey.description && <p className="text-gray-600">{survey.description}</p>}
        </div>
        
        {/* Progress indicator */}
        <div className="mb-6">
          <div className="flex justify-between mb-1">
            <span>Progresso</span>
            <span>{currentQuestionIndex + 1} de {survey.questions.length}</span>
          </div>
          <div className="w-full bg-gray-200 rounded-full h-2.5">
            <div 
              className="bg-blue-600 h-2.5 rounded-full" 
              style={{ width: `${((currentQuestionIndex + 1) / survey.questions.length) * 100}%` }}
            ></div>
          </div>
        </div>
        
        {/* Recording indicator */}
        {recording && (
          <div className="flex items-center mb-4 text-red-500">
            <div className="w-3 h-3 bg-red-500 rounded-full mr-2 animate-pulse"></div>
            <span>Gravando áudio...</span>
          </div>
        )}
        
        {/* Current question */}
        <div className="mb-6">
          <h2 className="text-xl font-semibold mb-4">
            {currentQuestionIndex + 1}. {currentQuestion.text}
            {currentQuestion.required && <span className="text-red-500 ml-1">*</span>}
          </h2>
          
          {/* Question options based on type */}
          {currentQuestion.type === 'single' && (
            <div className="space-y-2">
              {currentQuestion.options.map((option, optionIndex) => (
                <div key={optionIndex} className="flex items-center">
                  <input 
                    type="radio" 
                    id={`option-${currentQuestionIndex}-${optionIndex}`}
                    name={`question-${currentQuestionIndex}`}
                    value={option.value}
                    checked={answers[currentQuestionIndex].selectedOptions.includes(option.value)}
                    onChange={() => handleSingleChoice(currentQuestionIndex, option.value)}
                    className="mr-2"
                  />
                  <label htmlFor={`option-${currentQuestionIndex}-${optionIndex}`}>
                    {option.text}
                  </label>
                </div>
              ))}
            </div>
          )}
          
          {currentQuestion.type === 'multiple' && (

const PendingSurvey = require('../models/PendingSurvey');
const axios = require('axios');
const logger = require('../utils/logger');

// Process incoming messages
async function processMessage(client, message) {
  // Skip non-text messages and messages from groups
  if (!message.body || message.isGroupMsg) return;
  
  const chat = await message.getChat();
  const contact = await message.getContact();
  const phone = contact.number;
  
  logger.info(`Received message from ${phone}: ${message.body}`);
  
  // Check if there's an active survey for this user
  const pendingSurvey = await PendingSurvey.findOne({ 
    phone,
    completed: false
  }).sort({ startedAt: -1 });
  
  if (!pendingSurvey) {
    // No active survey, this might be a direct message
    await client.sendMessage(
      message.from,
      'Olá! Parece que você não possui nenhuma pesquisa ativa no momento. ' +
      'Se você recebeu um convite para uma pesquisa, por favor, aguarde que ela será enviada em breve.'
    );
    return;
  }
  
  // Handle INICIAR command
  if (pendingSurvey.currentQuestion === 0 && 
      message.body.trim().toUpperCase() === 'INICIAR') {
    return sendNextQuestion(client, pendingSurvey);
  }
  
  // Handle CANCELAR command
  if (message.body.trim().toUpperCase() === 'CANCELAR') {
    pendingSurvey.abandoned = true;
    await pendingSurvey.save();
    
    await client.sendMessage(
      message.from,
      'Pesquisa cancelada. Obrigado pelo seu tempo!'
    );
    return;
  }
  
  // Process answer for current question
  const currentQuestion = pendingSurvey.questions[pendingSurvey.currentQuestion - 1];
  if (!currentQuestion) return;
  
  const answer = message.body.trim();
  
  // Validate answer based on question type
  let isValidAnswer = false;
  let formattedAnswer = null;
  
  if (currentQuestion.type === 'open') {
    // Any answer is valid for open questions
    isValidAnswer = true;
    formattedAnswer = {
      question: currentQuestion._id,
      openResponse: answer,
      selectedOptions: []
    };
  } else if (['single', 'single_with_open'].includes(currentQuestion.type)) {
    // For single choice, answer should be a number corresponding to an option
    const optionNumber = parseInt(answer);
    if (!isNaN(optionNumber) && optionNumber > 0 && optionNumber <= currentQuestion.options.length) {
      isValidAnswer = true;
      formattedAnswer = {
        question: currentQuestion._id,
        selectedOptions: [currentQuestion.options[optionNumber - 1].value]
      };
      
      // If question requires open response, we need to ask for it
      if (currentQuestion.type === 'single_with_open') {
        pendingSurvey.pendingOpenResponse = true;
        await pendingSurvey.save();
        
        await client.sendMessage(
          message.from,
          'Por favor, explique sua resposta:'
        );
        return;
      }
    }
  } else if (['multiple', 'multiple_with_open'].includes(currentQuestion.type)) {
    // For multiple choice, answer should be comma-separated numbers
    const optionNumbers = answer.split(',').map(n => parseInt(n.trim())).filter(n => !isNaN(n));
    const validOptions = optionNumbers.filter(n => n > 0 && n <= currentQuestion.options.length);
    
    if (validOptions.length > 0) {
      isValidAnswer = true;
      formattedAnswer = {
        question: currentQuestion._id,
        selectedOptions: validOptions.map(n => currentQuestion.options[n - 1].value)
      };
      
      // If question requires open response, we need to ask for it
      if (currentQuestion.type === 'multiple_with_open') {
        pendingSurvey.pendingOpenResponse = true;
        await pendingSurvey.save();
        
        await client.sendMessage(
          message.from,
          'Por favor, explique sua resposta:'
        );
        return;
      }
    }
  }
  
  // Handle pending open response for single_with_open or multiple_with_open
  if (pendingSurvey.pendingOpenResponse) {
    pendingSurvey.pendingOpenResponse = false;
    pendingSurvey.answers[pendingSurvey.answers.length - 1].openResponse = answer;
    await pendingSurvey.save();
    
    return sendNextQuestion(client, pendingSurvey);
  }
  
  if (!isValidAnswer) {
    await client.sendMessage(
      message.from,
      'Desculpe, não entendi sua resposta. Por favor, responda conforme as instruções.'
    );
    return;
  }
  
  // Save the answer
  pendingSurvey.answers.push(formattedAnswer);
  pendingSurvey.currentQuestion++;
  await pendingSurvey.save();
  
  // Send next question or complete the survey
  await sendNextQuestion(client, pendingSurvey);
}

// Send the next question to the user
async function sendNextQuestion(client, survey) {
  if (survey.currentQuestion === 0) {
    // Starting the survey
    survey.currentQuestion = 1;
    await survey.save();
  }
  
  if (survey.currentQuestion > survey.questions.length) {
    // Survey completed
    survey.completed = true;
    survey.completedAt = new Date();
    await survey.save();
    
    // Submit the survey responses to the API
    try {
      await axios.post(`${process.env.API_URL}/api/surveys/${survey.surveyId}/responses/whatsapp`, {
        answers: survey.answers,
        phone: survey.phone,
        botId: process.env.BOT_ID,
        apiKey: process.env.API_KEY
      });
      
      await client.sendMessage(
        `${survey.phone}@c.us`,
        'Obrigado por completar a pesquisa! Suas respostas foram registradas com sucesso.'
      );
    } catch (error) {
      logger.error('Error submitting survey responses:', error);
      
      await client.sendMessage(
        `${survey.phone}@c.us`,
        'Obrigado por completar a pesquisa! ' +
        'Houve um problema ao registrar suas respostas, mas tentaremos novamente mais tarde.'
      );
    }
    
    return;
  }
  
  // Get the current question
  const question = survey.questions[survey.currentQuestion - 1];
  
  let questionText = `*Pergunta ${survey.currentQuestion} de ${survey.questions.length}*\n\n${question.text}`;
  
  // Add options for multiple-choice questions
  if (['single', 'multiple', 'single_with_open', 'multiple_with_open'].includes(question.type)) {
    questionText += '\n\n';
    
    question.options.forEach((option, index) => {
      questionText += `${index + 1}. ${option.text}\n`;
    });
    
    if (question.type === 'single' || question.type === 'single_with_open') {
      questionText += '\n_Responda com o número da opção escolhida._';
    } else {
      questionText += '\n_Responda com os números das opções escolhidas, separados por vírgula (ex: 1,3)._';
    }
  }
  
  // Send the question
  await client.sendMessage(`${survey.phone}@c.us`, questionText);
}

module.exports = { processMessage };

Conclusão
Este projeto implementa um sistema completo de pesquisa com as seguintes características principais:

Autenticação: Login por email/senha ou Google
Criação de Pesquisas: Interface intuitiva para criar pesquisas com filtros demográficos
Tipos de Perguntas: Suporte para escolha única, múltipla escolha e respostas abertas
Compartilhamento: Pesquisas podem ser compartilhadas por ID, link, email, SMS ou WhatsApp
Coleta de Dados: Captura de áudio e geolocalização durante as respostas
Dashboard: Visualização de resultados em tempo real
Sistema de Créditos: Pagamentos por PIX ou PayPal para envio de pesquisas
Integração WhatsApp: Bot para realização de pesquisas via chat
Administração: Painel de superadmin para gerenciar o sistema
Armazenamento Offline: Banco de dados local para funcionamento sem internet
